public static void saveToFile(Path path, Object data) throws IOException { Path normalized = path.toAbsolutePath().normalize(); ReentrantReadWriteLock.WriteLock writeLock = lockFor(normalized).writeLock(); writeLock.lock(); try { Path tempPath = Files.createTempFile(tempDir, normalized.getFileName() + ".", ".tmp"); Files.writeString(tempPath, toJson(data), StandardCharsets.UTF_8, StandardOpenOption.WRITE, StandardOpenOption.TRUNCATE_EXISTING); Files.move(tempPath, normalized, StandardCopyOption.ATOMIC_MOVE, StandardCopyOption.REPLACE_EXISTING); } finally { writeLock.unlock(); } }
